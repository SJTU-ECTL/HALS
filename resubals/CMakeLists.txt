# Set the minimum version of CMake that can be used
# To find the cmake version run
# $ cmake --version
cmake_minimum_required(VERSION 3.16)

# Set the project name
project(resubals)

option(HALS_ENABLE_TORCH "Enable the DATE 2027 LibTorch signed-LAC scorer" OFF)

# Set top directory
set(TOP_DIR ${PROJECT_SOURCE_DIR}/..)

# Collect source files
aux_source_directory(${PROJECT_SOURCE_DIR}/src sources)

find_package(OpenMP)
find_package(OpenSSL REQUIRED)
# Add include paths
include_directories(${TOP_DIR}/abc/src)
# include_directories(${TOP_DIR}/espresso/src)
include_directories(${PROJECT_SOURCE_DIR}/src)

# Add definitions
# add_definitions(-DLIN64 -g -Wall -O3 -std=c++2a -fopenmp)
add_definitions(-DLIN64 -g -Wall -O3 -std=c++2a)

# Keep the historical root executable by default while allowing clean-clone
# verification to place all generated files on a separate filesystem.  This
# matters for publication campaigns whose evidence volume can fill the source
# volume; an explicit CMAKE_RUNTIME_OUTPUT_DIRECTORY must not be ignored.
if(NOT CMAKE_RUNTIME_OUTPUT_DIRECTORY)
    set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${TOP_DIR})
endif()

# Add an executable
add_executable(resubals.out ${sources})
target_link_libraries(resubals.out libabc OpenSSL::Crypto -lm -ldl -rdynamic -lreadline -ltermcap -lpthread -lstdc++ -lrt -lstdc++fs -lboost_timer -lboost_system)
if(HALS_ENABLE_TORCH)
    find_package(Torch 2.5.1 EXACT REQUIRED)
    target_compile_definitions(resubals.out PRIVATE HALS_ENABLE_TORCH=1)
    target_link_libraries(resubals.out "${TORCH_LIBRARIES}")
    target_compile_options(resubals.out PRIVATE ${TORCH_CXX_FLAGS})

    # Dependency-light Python/C++ inference parity test.  Runtime arguments
    # supply a content-bound TorchScript model and graph tensor generated by
    # the Python reference, so the same executable is reusable in campaign CI.
    add_executable(
        signed_lac_scorer_test
        tests/signed_lac_scorer_test.cc
        src/signed_lac_scorer.cc
        src/transition_gmm.cc
    )
    target_include_directories(signed_lac_scorer_test PRIVATE
        ${PROJECT_SOURCE_DIR}/src)
    target_compile_definitions(signed_lac_scorer_test PRIVATE
        HALS_ENABLE_TORCH=1)
    target_compile_options(signed_lac_scorer_test PRIVATE ${TORCH_CXX_FLAGS})
    target_link_libraries(signed_lac_scorer_test
        "${TORCH_LIBRARIES}" OpenSSL::Crypto)
endif()
